git remote -v — показывает адреса удалённых репозиториев.

git tag — показывает список тегов.

git tag -a — создаёт аннотированный тег.

git show — показывает информацию о теге или коммите.

git push origin v1.0 — отправляет тег на GitHub.

git clone — копирует удалённый репозиторий.

git log --pretty=oneline — показывает краткую историю коммитов.

git checkout -b — создаёт ветку и переключается на неё.

git branch — показывает локальные ветки.

git switch — переключает между ветками.

git branch --all — показывает все ветки.

git switch - — возвращает на предыдущую ветку.

git branch -d — удаляет объединённую ветку.

git branch -D — принудительно удаляет ветку.
